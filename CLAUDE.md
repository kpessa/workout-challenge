### Store Management - Real-time with Optimistic Updates
```javascript
// Advanced store factory with error recovery
function createRealTimeStore() {
  const { subscribe, set, update } = writable(initialState);
  let unsubscribe = null;
  
  return {
    subscribe,
    initialize: async (userId) => {
      // Real-time listener with error handling
      unsubscribe = onSnapshot(
        collection(db, 'users', userId, 'workouts'),
        (snapshot) => {
          const data = snapshot.docs.map(doc => ({ id: doc.id, ...doc.data() }));
          set({ workouts: data, loading: false });
        },
        (error) => {
          console.error('Real-time listener error:', error);
          // Fallback to polling or cached data
        }
      );
    },
    addWorkout: async (data) => {
      const tempId = `temp_${Date.now()}`;
      // Optimistic update
      update(state => ({ 
        ...state, 
        workouts: [{ ...data, id: tempId }, ...state.workouts] 
      }));
      
      try {
        await addDoc(workoutsRef, data);
        // Real-time listener handles actual update
      } catch (error) {
        // Rollback optimistic update
        update(state => ({ 
          ...state, 
          workouts: state.workouts.filter(w => w.id !== tempId) 
        }));
        throw error;
      }
    },
    destroy: () => unsubscribe?.()
  };
}
```
