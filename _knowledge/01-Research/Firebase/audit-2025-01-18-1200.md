rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /users/{userId} {
      allow read: if request.auth != null && request.auth.uid == userId;
      
      match /workouts/{workoutId} {
        allow read, create: if request.auth != null 
          && request.auth.uid == userId
          && (request.method == 'create' ? isValidWorkout(request.resource.data) : true);
        allow update: if request.auth != null 
          && request.auth.uid == userId
          && resource.data.user_id == request.auth.uid
          && isValidWorkoutUpdate(request.resource.data);
        allow delete: if request.auth != null 
          && request.auth.uid == userId
          && resource.data.user_id == request.auth.uid;
      }
    }
  }
}

function isValidWorkout(data) {
  return data.keys().hasAll(['date', 'duration', 'type'])
    && data.duration is number
    && data.duration > 0
    && data.duration <= 1440  // Max 24 hours
    && data.date is string
    && data.date.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}$');
}

function isValidWorkoutUpdate(data) {
  return !data.keys().hasAny(['id', 'user_id', 'created_at'])  // Prevent modification of immutable fields
    && (data.keys().hasAny(['date', 'duration', 'type', 'updated_at']));
}

  async get<T>(key: string, fetcher: () => Promise<T>): Promise<T> {

## Firebase Pattern Templates

### Real-time Store Factory Pattern

```typescript
// Reusable real-time store pattern
function createRealtimeFirestoreStore<T>(
  collectionPath: (userId: string) => string,
  queryModifiers?: (q: Query) => Query,
  validator?: (data: any) => T
) {
  const { subscribe, set, update } = writable<{
    data: T[];
    loading: boolean;
    error: string | null;
    syncing: boolean;
  }>({
    data: [],
    loading: true,
    error: null,
    syncing: false
  });
  
  let unsubscribe: Unsubscribe | null = null;
  
  function initialize(userId: string) {
    if (unsubscribe) unsubscribe();
    
    const path = collectionPath(userId);
    let q = query(collection(db, path));
    
    if (queryModifiers) {
      q = queryModifiers(q);
    }
    
    unsubscribe = onSnapshot(
      q,
      (snapshot) => {
        const data = snapshot.docs.map(doc => {
          const item = { id: doc.id, ...doc.data() };
          return validator ? validator(item) : item;
        });
        update(state => ({ ...state, data, loading: false, error: null }));
      },
      (error) => {
        const errorInfo = handleFirestoreError(error);
        update(state => ({ ...state, loading: false, error: errorInfo.message }));
      }
    );
  }
  
  return { subscribe, initialize, destroy: () => unsubscribe?.() };
}

// Usage example:
export const workoutTypesRealtime = createRealtimeFirestoreStore<WorkoutType>(
  (userId) => `users/${userId}/workout_types`,
  (q) => orderBy(q, 'name', 'asc'),
  (data) => validateWorkoutType(data)
);
```

### Optimistic Update Pattern

```typescript
// Advanced optimistic update with conflict resolution
export async function createOptimisticOperation<T, U>(
  operation: () => Promise<T>,
  optimisticUpdate: (state: U) => U,
  rollback: (state: U, error: any) => U,
  store: { update: (fn: (state: U) => U) => void }
): Promise<T> {
  // Apply optimistic update
  store.update(optimisticUpdate);
  
  try {
    const result = await operation();
    return result;
  } catch (error) {
    // Rollback on failure
    store.update(state => rollback(state, error));
    throw error;
  }
}
```
