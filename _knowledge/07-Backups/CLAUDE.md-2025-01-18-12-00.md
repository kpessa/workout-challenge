## 🔧 Development Guidelines (Updated)

### Store Management - Advanced Pattern
```javascript
// Store factory with real-time capabilities and optimistic updates
function createStoreFactory() {
  const { subscribe, set, update } = writable(initialState);
  let unsubscribe = null;
  
  return {
    subscribe,
    initialize: async () => { /* real-time listener setup */ },
    customMethod: async (data) => { 
      // Optimistic update
      update(state => ({ ...state, optimistic: data }));
      try {
        await saveToFirebase(data);
      } catch (error) {
        // Rollback optimistic update
        update(state => revertOptimistic(state));
      }
    },
    destroy: () => { /* cleanup listeners */ }
  };
}
```

### Firebase Operations - Real-time Pattern
```javascript
import { db, auth } from '$lib/services/firebase';
import { onSnapshot } from 'firebase/firestore';

// Preferred: Real-time listeners over polling
const unsubscribe = onSnapshot(
  collection(db, 'users', userId, 'workouts'),
  (snapshot) => {
    // Handle real-time updates
  }
);
```
